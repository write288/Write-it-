                <div class="color-dot active" style="background: #2f3542;" onclick="changeColor('#2f3542', this)"></div>
                <div class="color-dot" style="background: #ff4757;" onclick="changeColor('#ff4757', this)"></div>
                <div class="color-dot" style="background: #2ed573;" onclick="changeColor('#2ed573', this)"></div>
                <div class="color-dot" style="background: #1e90ff;" onclick="changeColor('#1e90ff', this)"></div>
                <div class="color-dot" style="background: #ffa502;" onclick="changeColor('#ffa502', this)"></div>
            </div>
            <button class="btn btn-clear" onclick="clearCanvas()">Clear</button>
        </div>
        <div class="row">
            <span style="color: white; font-size: 12px; min-width: 65px;">Size: <span id="sizeVal">5</span>px</span>
            <input type="range" id="brushSize" min="2" max="30" value="5" oninput="changeSize(this.value)">
        </div>
    </div>

    <script>
        const canvas = document.getElementById('paintCanvas');
        const ctx = canvas.getContext('2d');
        const sizeVal = document.getElementById('sizeVal');
        
        let isDrawing = false;
        let currentScale = 1;

        // Automatically configure dimensions to fit the precise device screen
        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            
            // Re-apply baseline path preferences after clear/resize loops
            ctx.lineCap = 'round';
            ctx.lineJoin = 'round';
            ctx.lineWidth = document.getElementById('brushSize').value;
            // Retain currently active selection color properties
            const activeColor = document.querySelector('.color-dot.active').style.backgroundColor;
            ctx.strokeStyle = activeColor;
        }

        function changeColor(color, element) {
            ctx.strokeStyle = color;
            document.querySelectorAll('.color-dot').forEach(dot => dot.classList.remove('active'));
            element.classList.add('active');
        }

        function changeSize(size) {
            ctx.lineWidth = size;
            sizeVal.innerText = size;
        }

        function clearCanvas() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
        }

        // Helper calculations tracking precision touch trajectories
        function getTouchCoordinates(e) {
            const touch = e.touches[0];
            return { x: touch.clientX, y: touch.clientY };
        }

        // Hardware-optimized touch event intercept hooks
        canvas.addEventListener('touchstart', (e) => {
            isDrawing = true;
            const pos = getTouchCoordinates(e);
            ctx.beginPath();
            ctx.moveTo(pos.x, pos.y);
            e.preventDefault(); 
        }, { passive: false });

        canvas.addEventListener('touchmove', (e) => {
            if (!isDrawing) return;
            const pos = getTouchCoordinates(e);
            ctx.lineTo(pos.x, pos.y);
            ctx.stroke();
            e.preventDefault();
        }, { passive: false });

        canvas.addEventListener('touchend', () => {
            isDrawing = false;
            ctx.closePath();
        });

        // Initialize display configuration layout rules
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();
    </script>
